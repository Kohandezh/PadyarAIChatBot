#!/bin/sh
# Download the bench models in spec order; resumable (curl -C -), then sha256.
set -u
cd /var/lib/padyar/llm/models
for spec in \
  "unsloth/gemma-4-26B-A4B-it-GGUF gemma-4-26B-A4B-it-UD-Q4_K_M.gguf" \
  "unsloth/gemma-4-12b-it-GGUF gemma-4-12b-it-Q4_K_M.gguf" \
  "Qwen/Qwen3-14B-GGUF Qwen3-14B-Q4_K_M.gguf"; do
  set -- $spec
  echo "$(date -Is) start $1/$2"
  curl -sSL --retry 20 --retry-delay 5 -C - -o "$2" "https://huggingface.co/$1/resolve/main/$2"
  echo "$(date -Is) done rc=$? $2 size=$(stat -c %s "$2")"
  sha256sum "$2" | tee -a SHA256SUMS
done
echo "$(date -Is) ALL DONE"
